## Steps

### Step 1: Create a Basic Role

Create a simple role structure:

```bash
# Create role directory structure
mkdir -p /tmp/roles/webserver/{tasks,handlers,templates,files,vars,defaults,meta}

# Create role tasks
cat > /tmp/roles/webserver/tasks/main.yml << 'EOF'
---
# Main tasks for webserver role
- name: Display role information
  debug:
    msg: "Installing and configuring webserver"

- name: Create webserver directory
  file:
    path: /tmp/webserver
    state: directory
    mode: '0755'

- name: Install web application (simulated)
  copy:
    content: |
      #!/bin/bash
      echo "Web server is running on port {{ http_port }}"
    dest: /tmp/webserver/app.sh
    mode: '0755'

- name: Deploy configuration
  template:
    src: webserver.conf.j2
    dest: /tmp/webserver/webserver.conf
  notify: restart webserver

- name: Create index page
  copy:
    src: index.html
    dest: /tmp/webserver/index.html
EOF

# Create role handlers
cat > /tmp/roles/webserver/handlers/main.yml << 'EOF'
---
# Handlers for webserver role
- name: restart webserver
  debug:
    msg: "Webserver would be restarted here"
EOF

# Create role template
cat > /tmp/roles/webserver/templates/webserver.conf.j2 << 'EOF'
# Webserver Configuration
# Generated by Ansible

[server]
port={{ http_port }}
host={{ server_host }}
document_root={{ document_root }}
max_connections={{ max_connections }}

[logging]
log_level={{ log_level }}
access_log={{ access_log }}
error_log={{ error_log }}
EOF

# Create role files
cat > /tmp/roles/webserver/files/index.html << 'EOF'
<!DOCTYPE html>
<html>
<head>
    <title>Welcome</title>
</head>
<body>
    <h1>Welcome to the Web Server!</h1>
    <p>This page was deployed by Ansible.</p>
</body>
</html>
EOF

# Create role defaults
cat > /tmp/roles/webserver/defaults/main.yml << 'EOF'
---
# Default variables for webserver role
http_port: 8080
server_host: 0.0.0.0
document_root: /tmp/webserver
max_connections: 100
log_level: info
access_log: /tmp/webserver/access.log
error_log: /tmp/webserver/error.log
EOF

# Create role metadata
cat > /tmp/roles/webserver/meta/main.yml << 'EOF'
---
galaxy_info:
  author: Ansible Labs
  description: Basic webserver role
  license: MIT
  min_ansible_version: 2.9
  platforms:
    - name: Ubuntu
      versions:
        - focal
        - jammy

dependencies: []
EOF
```

### Step 2: Use the Role in a Playbook

```bash
cat > use-role.yml << 'EOF'
---
- name: Using Webserver Role
  hosts: localhost
  connection: local
  roles:
    - role: webserver
      vars:
        http_port: 8080
        log_level: debug
EOF

# Run the playbook
ansible-playbook use-role.yml
```

### Step 3: Create Multiple Roles

Create additional roles for a complete application stack:

```bash
# Database role
mkdir -p /tmp/roles/database/{tasks,handlers,templates,defaults,meta}

cat > /tmp/roles/database/tasks/main.yml << 'EOF'
---
- name: Create database directory
  file:
    path: /tmp/database
    state: directory

- name: Deploy database configuration
  template:
    src: database.conf.j2
    dest: /tmp/database/database.conf
  notify: restart database

- name: Initialize database
  copy:
    content: |
      # Database initialized
      version={{ db_version }}
      port={{ db_port }}
    dest: /tmp/database/init.sql
EOF

cat > /tmp/roles/database/handlers/main.yml << 'EOF'
---
- name: restart database
  debug:
    msg: "Database would be restarted"
EOF

cat > /tmp/roles/database/templates/database.conf.j2 << 'EOF'
# Database Configuration
[database]
port={{ db_port }}
max_connections={{ db_max_connections }}
data_directory={{ db_data_dir }}
EOF

cat > /tmp/roles/database/defaults/main.yml << 'EOF'
---
db_port: 5432
db_max_connections: 200
db_data_dir: /tmp/database/data
db_version: "14.0"
EOF

cat > /tmp/roles/database/meta/main.yml << 'EOF'
---
galaxy_info:
  author: Ansible Labs
  description: Database role
  license: MIT
  min_ansible_version: 2.9

dependencies: []
EOF

# Application role
mkdir -p /tmp/roles/application/{tasks,handlers,templates,defaults,vars}

cat > /tmp/roles/application/tasks/main.yml << 'EOF'
---
- name: Create application directory
  file:
    path: "{{ app_directory }}"
    state: directory

- name: Deploy application
  template:
    src: app.py.j2
    dest: "{{ app_directory }}/app.py"
    mode: '0755'

- name: Create application config
  template:
    src: config.ini.j2
    dest: "{{ app_directory }}/config.ini"
  notify: restart application
EOF

cat > /tmp/roles/application/handlers/main.yml << 'EOF'
---
- name: restart application
  debug:
    msg: "Application would be restarted"
EOF

cat > /tmp/roles/application/templates/app.py.j2 << 'EOF'
#!/usr/bin/env python3
# {{ app_name }} - Version {{ app_version }}

def main():
    print("{{ app_name }} is running on port {{ app_port }}")
    print("Database: {{ database_host }}:{{ database_port }}")

if __name__ == "__main__":
    main()
EOF

cat > /tmp/roles/application/templates/config.ini.j2 << 'EOF'
[application]
name={{ app_name }}
version={{ app_version }}
port={{ app_port }}

[database]
host={{ database_host }}
port={{ database_port }}
EOF

cat > /tmp/roles/application/defaults/main.yml << 'EOF'
---
app_name: MyApplication
app_version: 1.0.0
app_port: 3000
app_directory: /tmp/application
database_host: localhost
database_port: 5432
EOF

cat > /tmp/roles/application/meta/main.yml << 'EOF'
---
galaxy_info:
  author: Ansible Labs
  description: Application role
  license: MIT

dependencies: []
EOF
```

### Step 4: Use Multiple Roles


### Step 5: Role Dependencies

Create a role with dependencies:

```bash
# Create monitoring role
mkdir -p /tmp/roles/monitoring/{tasks,defaults,meta}

cat > /tmp/roles/monitoring/tasks/main.yml << 'EOF'
---
- name: Setup monitoring
  copy:
    content: |
      # Monitoring Configuration
      targets:
        - webserver:{{ http_port }}
        - database:{{ db_port }}
    dest: /tmp/monitoring-config.yml
EOF

cat > /tmp/roles/monitoring/defaults/main.yml << 'EOF'
---
monitoring_enabled: true
EOF

# Add dependencies to monitoring role
cat > /tmp/roles/monitoring/meta/main.yml << 'EOF'
---
galaxy_info:
  author: Ansible Labs
  description: Monitoring role

dependencies:
  - role: webserver
  - role: database
EOF

# Use role with dependencies
cat > role-dependencies.yml << 'EOF'
---
- name: Deploy with Dependencies
  hosts: localhost
  connection: local
  
  roles:
    - monitoring
EOF

ansible-playbook role-dependencies.yml
```

### Step 6: Role with Pre and Post Tasks

```bash
cat > role-with-tasks.yml << 'EOF'
---
- name: Deployment with Pre and Post Tasks
  hosts: localhost
  connection: local
  
  pre_tasks:
    - name: Pre-deployment check
      debug:
        msg: "Starting deployment process"
    
    - name: Create backup directory
      file:
        path: /tmp/backup
        state: directory
  
  roles:
    - role: application
      vars:
        app_name: ProductionApp
  
  post_tasks:
    - name: Verify deployment
      stat:
        path: /tmp/application/app.py
      register: app_file
    
    - name: Display deployment status
      debug:
        msg: "Deployment {{ 'successful' if app_file.stat.exists else 'failed' }}"
    
    - name: Create deployment marker
      copy:
        content: |
          Deployment completed: {{ ansible_date_time.iso8601 }}
          Application: {{ app_name }}
        dest: /tmp/deployment-complete.txt
EOF

ansible-playbook role-with-tasks.yml
```

### Step 7: Role Variables Organization

```bash
# Create a role with organized variables
mkdir -p /tmp/roles/webapp/{tasks,vars,defaults}

cat > /tmp/roles/webapp/tasks/main.yml << 'EOF'
---
- name: Load environment-specific variables
  include_vars: "{{ item }}"
  with_first_found:
    - "{{ environment }}.yml"
    - "default.yml"

- name: Display loaded configuration
  debug:
    msg: |
      Environment: {{ environment }}
      Debug: {{ debug_enabled }}
      Workers: {{ worker_count }}

- name: Deploy with environment config
  copy:
    content: |
      Environment: {{ environment }}
      Debug: {{ debug_enabled }}
      Workers: {{ worker_count }}
      Database: {{ db_connection_string }}
    dest: "/tmp/webapp-{{ environment }}.conf"
EOF

cat > /tmp/roles/webapp/vars/production.yml << 'EOF'
---
debug_enabled: false
worker_count: 8
db_connection_string: "postgresql://prod-db:5432/webapp"
EOF

cat > /tmp/roles/webapp/vars/development.yml << 'EOF'
---
debug_enabled: true
worker_count: 2
db_connection_string: "postgresql://localhost:5432/webapp_dev"
EOF

cat > /tmp/roles/webapp/vars/default.yml << 'EOF'
---
debug_enabled: false
worker_count: 4
db_connection_string: "postgresql://localhost:5432/webapp"
EOF

cat > /tmp/roles/webapp/defaults/main.yml << 'EOF'
---
environment: development
EOF

# Use the role with different environments
cat > webapp-deployment.yml << 'EOF'
---
- name: Deploy WebApp - Development
  hosts: localhost
  connection: local
  roles:
    - role: webapp
      vars:
        environment: development

- name: Deploy WebApp - Production
  hosts: localhost
  connection: local
  roles:
    - role: webapp
      vars:
        environment: production
EOF

ansible-playbook webapp-deployment.yml
```

### Step 8: Role Conditionals and Tags

```bash
cat > /tmp/roles/conditional-role/tasks/main.yml << 'EOF'
---
- name: Task for production only
  debug:
    msg: "Running production configuration"
  when: environment == "production"
  tags: production

- name: Task for development only
  debug:
    msg: "Running development configuration"
  when: environment == "development"
  tags: development

- name: Always run this task
  debug:
    msg: "This runs in all environments"
  tags: always

- name: Optional feature
  debug:
    msg: "Feature enabled"
  when: enable_feature | default(false)
  tags: optional
EOF

mkdir -p /tmp/roles/conditional-role/defaults
cat > /tmp/roles/conditional-role/defaults/main.yml << 'EOF'
---
environment: development
enable_feature: false
EOF

cat > conditional-roles.yml << 'EOF'
---
- name: Conditional Role Execution
  hosts: localhost
  connection: local
  
  roles:
    - role: conditional-role
      vars:
        environment: production
        enable_feature: true
EOF

# Run entire playbook
ansible-playbook conditional-roles.yml

# Run only production tasks
ansible-playbook conditional-roles.yml --tags production

# Skip optional tasks
ansible-playbook conditional-roles.yml --skip-tags optional
```

### Step 9: Role Import vs Include

```bash
cat > import-include-roles.yml << 'EOF'
---
- name: Import vs Include Roles
  hosts: localhost
  connection: local
  
  tasks:
    - name: Static import (processed at playbook parse time)
      import_role:
        name: webserver
      vars:
        http_port: 8081
    
    - name: Dynamic include (processed at runtime)
      include_role:
        name: application
      vars:
        app_name: DynamicApp
      when: deploy_app | default(true)
    
    - name: Include role with tags
      include_role:
        name: database
      tags: database
EOF

ansible-playbook import-include-roles.yml
```

### Step 10: Complete Role Example

```bash
# Create a comprehensive role
mkdir -p /tmp/roles/complete-app/{tasks,handlers,templates,files,vars,defaults,meta}

cat > /tmp/roles/complete-app/tasks/main.yml << 'EOF'
---
- name: Include OS-specific variables
  include_vars: "{{ ansible_os_family }}.yml"
  ignore_errors: yes

- name: Create application structure
  file:
    path: "{{ item }}"
    state: directory
  loop:
    - "{{ app_root }}"
    - "{{ app_root }}/logs"
    - "{{ app_root }}/data"
    - "{{ app_root }}/config"

- name: Deploy application files
  copy:
    src: "{{ item }}"
    dest: "{{ app_root }}/"
  with_fileglob:
    - "files/*"

- name: Deploy configuration
  template:
    src: application.conf.j2
    dest: "{{ app_root }}/config/application.conf"
  notify:
    - restart application
    - notify admin

- name: Create systemd service file (simulated)
  template:
    src: service.j2
    dest: /tmp/complete-app.service

- name: Display deployment summary
  debug:
    msg: |
      Application deployed:
      - Root: {{ app_root }}
      - Port: {{ app_port }}
      - Environment: {{ app_environment }}
EOF

cat > /tmp/roles/complete-app/handlers/main.yml << 'EOF'
---
- name: restart application
  debug:
    msg: "Application restarted"

- name: notify admin
  debug:
    msg: "Admin notified of configuration change"
EOF

cat > /tmp/roles/complete-app/templates/application.conf.j2 << 'EOF'
# Complete Application Configuration
[app]
name={{ app_name }}
version={{ app_version }}
environment={{ app_environment }}
port={{ app_port }}

[paths]
root={{ app_root }}
logs={{ app_root }}/logs
data={{ app_root }}/data

[features]
{% for feature, enabled in app_features.items() %}
{{ feature }}={{ enabled | lower }}
{% endfor %}
EOF

cat > /tmp/roles/complete-app/templates/service.j2 << 'EOF'
[Unit]
Description={{ app_name }} Service
After=network.target

[Service]
Type=simple
User={{ app_user }}
WorkingDirectory={{ app_root }}
ExecStart={{ app_root }}/start.sh
Restart=on-failure

[Install]
WantedBy=multi-user.target
EOF

cat > /tmp/roles/complete-app/defaults/main.yml << 'EOF'
---
app_name: CompleteApp
app_version: 1.0.0
app_environment: production
app_port: 3000
app_root: /tmp/complete-app
app_user: "{{ ansible_user_id }}"

app_features:
  caching: true
  monitoring: true
  debugging: false
EOF

cat > /tmp/roles/complete-app/meta/main.yml << 'EOF'
---
galaxy_info:
  author: Ansible Labs
  description: Complete application deployment role
  license: MIT
  min_ansible_version: 2.9
  
dependencies: []

allow_duplicates: no
EOF

# Use the complete role
cat > complete-deployment.yml << 'EOF'
---
- name: Complete Application Deployment
  hosts: localhost
  connection: local
  
  roles:
    - role: complete-app
      vars:
        app_environment: production
        app_port: 8000
        app_features:
          caching: true
          monitoring: true
          debugging: false
EOF

ansible-playbook complete-deployment.yml
```

## Role Directory Structure

```
roles/
└── role_name/
    ├── tasks/
    │   └── main.yml
    ├── handlers/
    │   └── main.yml
    ├── templates/
    │   └── template.j2
    ├── files/
    │   └── file.txt
    ├── vars/
    │   └── main.yml
    ├── defaults/
    │   └── main.yml
    ├── meta/
    │   └── main.yml
    ├── library/
    ├── module_utils/
    └── lookup_plugins/
```

## Troubleshooting

### Problem: Role not found

**Solution:** Check roles path
```bash
ansible-config dump | grep ROLES_PATH
export ANSIBLE_ROLES_PATH=/tmp/roles
```

### Problem: Variables not loading

**Solution:** Check variable precedence (defaults < vars < playbook vars)

### Problem: Role dependencies not running

**Solution:** Ensure meta/main.yml is properly configured
